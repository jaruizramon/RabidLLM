<script>
(() => {
	const state = { artifacts: [], selectedPath: null, loading: false, lastFetched: 0 };
	const $ = id => document.getElementById(id);
	const listEl = $("list");
	const emptyEl = $("empty");
	const viewerEl = $("viewer");
	const surfaceEl = $("surface");
	const titleEl = $("title");
	const pathEl = $("path");
	const factsEl = $("facts");
	const countEl = $("count");
	const cwdEl = $("cwd");
	const openEl = $("open");
	const reloadEl = $("reload");
	const copyEl = $("copy");
	const refreshEl = $("refresh");
	let toastTimer = 0;

	function toast(message, error = false) {
		const el = $("toast");
		el.textContent = message;
		el.classList.toggle("error", error);
		el.classList.add("show");
		clearTimeout(toastTimer);
		toastTimer = setTimeout(() => el.classList.remove("show"), 2600);
	}

	function formatBytes(bytes) {
		if (!Number.isFinite(bytes) || bytes < 0) return "—";
		if (bytes < 1024) return bytes + " B";
		const units = ["KB", "MB", "GB"];
		let value = bytes / 1024;
		let unit = 0;
		while (value >= 1024 && unit < units.length - 1) { value /= 1024; unit++; }
		return value.toFixed(value >= 10 ? 0 : 1) + " " + units[unit];
	}

	function formatTime(ms) {
		const date = new Date(ms);
		return date.toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" });
	}

	function itemFor(path) {
		return state.artifacts.find(artifact => artifact.path === path);
	}

	function fileUrl(artifact) {
		return "/file?path=" + encodeURIComponent(artifact.path) + "&v=" + artifact.mtimeMs;
	}

	function previewUrl(artifact, refresh) {
		return "/api/preview?path=" + encodeURIComponent(artifact.path) + "&v=" + artifact.mtimeMs + (refresh ? "&refresh=1" : "");
	}

	function kindGlyph(kind) {
		if (kind === "webpage") return "▤";
		if (kind === "svg") return "◈";
		return "▧";
	}

	function renderList() {
		listEl.textContent = "";
		for (const artifact of state.artifacts) {
			const item = document.createElement("div");
			item.className = "item" + (artifact.path === state.selectedPath ? " selected" : "");
			item.tabIndex = 0;
			item.dataset.path = artifact.path;
			item.addEventListener("click", () => select(artifact.path));

			const thumb = document.createElement("div");
			thumb.className = "thumb";
			if (artifact.kind === "image" || artifact.kind === "svg") {
				const img = document.createElement("img");
				img.loading = "lazy";
				img.alt = "";
				img.src = fileUrl(artifact);
				thumb.append(img);
			} else {
				const glyph = document.createElement("span");
				glyph.className = "glyph";
				glyph.textContent = kindGlyph(artifact.kind);
				thumb.append(glyph);
			}

			const text = document.createElement("div");
			text.className = "text";
			const name = document.createElement("div");
			name.className = "name";
			name.textContent = artifact.name;
			const sub = document.createElement("div");
			sub.className = "sub";
			const badge = document.createElement("span");
			badge.className = "badge " + artifact.kind;
			badge.textContent = artifact.kind === "webpage" ? "web" : artifact.kind;
			sub.append(badge, document.createTextNode("  " + formatBytes(artifact.size)));
			text.append(name, sub);
			item.append(thumb, text);
			listEl.append(item);
		}
	}

	function select(path) {
		if (state.selectedPath === path) return;
		state.selectedPath = path;
		renderList();
		renderDetail(false);
	}

	function renderDetail(refresh) {
		const artifact = itemFor(state.selectedPath);
		emptyEl.hidden = true;
		viewerEl.hidden = !artifact;
		if (!artifact) { emptyEl.hidden = false; return; }
		titleEl.textContent = artifact.title || artifact.name;
		pathEl.textContent = artifact.path;
		factsEl.textContent = "";
		const facts = [
			["kind", artifact.kind],
			["size", formatBytes(artifact.size)],
			["modified", formatTime(artifact.mtimeMs)],
			["source", artifact.tool || artifact.source],
		];
		for (const [key, value] of facts) {
			const span = document.createElement("span");
			const label = document.createElement("b");
			label.textContent = value;
			span.append(key + " ", label);
			factsEl.append(span);
		}
		surfaceEl.textContent = "";
		const loading = document.createElement("div");
		loading.className = "placeholder";
		const spinner = document.createElement("div");
		spinner.className = "spinner";
		loading.append(spinner, document.createTextNode("Rendering preview…"));
		surfaceEl.append(loading);
		reloadEl.hidden = artifact.kind !== "webpage";
		openEl.textContent = artifact.kind === "webpage" ? "Open page" : "Open image";

		const img = document.createElement("img");
		img.alt = artifact.title || artifact.name;
		img.className = artifact.kind === "webpage" ? "shot" : "art";
		img.src = artifact.kind === "webpage" ? previewUrl(artifact, refresh) : fileUrl(artifact);
		img.addEventListener("load", () => { if (surfaceEl.contains(loading)) { surfaceEl.textContent = ""; surfaceEl.append(img); } });
		img.addEventListener("error", () => {
			surfaceEl.textContent = "";
			const message = document.createElement("div");
			message.className = "placeholder";
			message.textContent = artifact.kind === "webpage"
				? "Could not render this webpage. Open it in a browser window instead."
				: "Could not load this image.";
			surfaceEl.append(message);
			toast(artifact.kind === "webpage" ? "Webpage preview failed" : "Image preview failed", true);
		});
	}

	async function refreshArtifacts(force) {
		if (state.loading) return;
		state.loading = true;
		refreshEl.disabled = true;
		try {
			const response = await fetch("/api/artifacts" + (force ? "?refresh=1" : ""), { cache: "no-store" });
			if (!response.ok) throw new Error("HTTP " + response.status);
			const data = await response.json();
			const previous = itemFor(state.selectedPath);
			state.artifacts = Array.isArray(data.artifacts) ? data.artifacts : [];
			state.lastFetched = Date.now();
			countEl.textContent = String(state.artifacts.length);
			cwdEl.textContent = data.cwd ? " · " + data.cwd : "";
			if (!itemFor(state.selectedPath)) state.selectedPath = state.artifacts[0]?.path || null;
			renderList();
			const current = itemFor(state.selectedPath);
			const changed = !previous || !current || previous.mtimeMs !== current.mtimeMs || previous.kind !== current.kind;
			renderDetail(changed);
		} catch (error) {
			toast("Could not load artifacts: " + (error.message || error), true);
		} finally {
			state.loading = false;
			refreshEl.disabled = false;
		}
	}

	async function openSelected() {
		const artifact = itemFor(state.selectedPath);
		if (!artifact) return;
		if (artifact.kind === "webpage") {
			try {
				const response = await fetch("/api/open", {
					method: "POST",
					headers: { "content-type": "application/json" },
					body: JSON.stringify({ path: artifact.path }),
				});
				if (!response.ok) throw new Error("HTTP " + response.status);
				toast("Opening in your browser…");
			} catch (error) {
				toast("Could not open the page: " + (error.message || error), true);
			}
			return;
		}
		window.open(fileUrl(artifact), "_blank", "noopener,noreferrer");
	}

	async function copySelected() {
		const artifact = itemFor(state.selectedPath);
		if (!artifact) return;
		try {
			await navigator.clipboard.writeText(artifact.path);
			toast("Copied path");
		} catch {
			const area = document.createElement("textarea");
			area.value = artifact.path;
			document.body.append(area);
			area.select();
			document.execCommand("copy");
			area.remove();
			toast("Copied path");
		}
	}

	refreshEl.addEventListener("click", () => refreshArtifacts(true));
	openEl.addEventListener("click", openSelected);
	reloadEl.addEventListener("click", () => renderDetail(true));
	copyEl.addEventListener("click", copySelected);
	document.addEventListener("keydown", event => {
		if (event.target instanceof HTMLInputElement || event.target instanceof HTMLTextAreaElement) return;
		const index = state.artifacts.findIndex(artifact => artifact.path === state.selectedPath);
		if (event.key === "ArrowDown" || event.key === "j") {
			event.preventDefault();
			const next = state.artifacts[Math.min(state.artifacts.length - 1, index + 1)];
			if (next) select(next.path);
		} else if (event.key === "ArrowUp" || event.key === "k") {
			event.preventDefault();
			const next = state.artifacts[Math.max(0, index - 1)];
			if (next) select(next.path);
		} else if (event.key === "Enter" || event.key === "o") {
			event.preventDefault();
			openSelected();
		} else if (event.key === "r") {
			event.preventDefault();
			refreshArtifacts(true);
		} else if (event.key === "c") {
			event.preventDefault();
			copySelected();
		}
	});

	refreshArtifacts(true);
	setInterval(() => {
		if (!document.hidden) refreshArtifacts(false);
	}, 2500);
	document.addEventListener("visibilitychange", () => {
		if (!document.hidden && Date.now() - state.lastFetched > 2500) refreshArtifacts(false);
	});
})();
</script>
